function replacePetSvgs(filePath, newJsCode, cssCode) {
    let content = fs.readFileSync(filePath, 'utf8');

    // Replace PET_SVGS block
    const targetStart = 'const PET_SVGS = {';
    const startIdx = content.indexOf(targetStart);
    if (startIdx === -1) {
        console.error(`Could not find PET_SVGS in ${filePath}`);
        return;
    }

    // Find the matching end index
    let openBrackets = 1;
    let endIdx = -1;
    for (let i = startIdx + targetStart.length; i < content.length; i++) {
        if (content[i] === '{') openBrackets++;
        else if (content[i] === '}') openBrackets--;

        if (openBrackets === 0) {
            // Find the trailing semicolon if present
            if (content[i + 1] === ';') {
                endIdx = i + 2;
            } else {
                endIdx = i + 1;
            }
            break;
        }
    }

    if (endIdx === -1) {
        console.error(`Could not find closing bracket for PET_SVGS in ${filePath}`);
        return;
    }

    const before = content.substring(0, startIdx);
    const after = content.substring(endIdx);
    content = before + newJsCode + after;

    // Insert CSS before </style>
    const styleEndIdx = content.indexOf('</style>');
    if (styleEndIdx === -1) {
        console.error(`Could not find </style> in ${filePath}`);
        return;
    }

    const cssBefore = content.substring(0, styleEndIdx);
    const cssAfter = content.substring(styleEndIdx);
    content = cssBefore + cssCode + cssAfter;

    fs.writeFileSync(filePath, content, 'utf8');
    console.log(`Successfully updated ${filePath}`);
}

const fs = require('fs');

function searchDir(dir) {
    const files = fs.readdirSync(dir);
    files.forEach(file => {
        const filePath = path.join(dir, file);
        const stat = fs.statSync(filePath);
        if (stat.isDirectory()) {
            if (file !== 'node_modules' && file !== '.git') {
                searchDir(filePath);
            }
        } else if (stat.isFile() && (file.endsWith('.html') || file.endsWith('.js') || file.endsWith('.css') || file.endsWith('.md'))) {
            const content = fs.readFileSync(filePath, 'utf8');
            const queries = ['히스토리', '오늘의 고민과 명언', '기록된 히스토리', 'worry'];
            queries.forEach(q => {
                if (content.includes(q)) {
                    console.log(`Found "${q}" in ${filePath}`);
                }
            });
        }
    });
}

function post(url, data) {
    return new Promise((resolve, reject) => {
        const parsedUrl = new URL(url);
        const options = {
            hostname: parsedUrl.hostname,
            port: parsedUrl.port,
            path: parsedUrl.pathname,
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            }
        };

        const req = http.request(options, (res) => {
            let body = '';
            res.on('data', (chunk) => { body += chunk; });
            res.on('end', () => {
                resolve({
                    statusCode: res.statusCode,
                    body: body
                });
            });
        });

        req.on('error', reject);
        req.write(JSON.stringify(data));
        req.end();
    });
}
